Double-Entry Ledger System (F002)
File: internal/wallet/ledger.go

Key Principle:

Setiap transaksi finansial menciptakan 2 entries: DEBIT dan CREDIT dengan amount yang identik.
Invariant: SUM(debit) = SUM(credit) untuk setiap reference_id.

Ledger Entries Structure:

golang
type LedgerEntry struct {
    ID              uuid.UUID         // Primary key
    WalletID        uuid.UUID         // Wallet yang berpengaruh
    EntryType       string            // ENUM: DEBIT | CREDIT
    Amount          decimal.Decimal   // Financial precision (2 decimals)
    BalanceAfter    decimal.Decimal   // Snapshot balance setelah entry
    ReferenceID     uuid.UUID         // Link ke transaction (topup_id, transfer_id)
    ReferenceType   string            // WALLET_TOPUP | WALLET_TRANSFER | RIDE_SETTLEMENT
    Description     string            // Human-readable (e.g., "Top-up via simulated payment")
    IsReversed      bool              // Flag untuk reversal (tidak delete)
    CreatedAt       time.Time         // Immutable timestamp
}

Token Blacklist Service — FIXED:

golang
type TokenBlacklistService struct {
    redis *redis.Client
}

// Blacklist token on logout or suspend
func (s *TokenBlacklistService) BlacklistToken(ctx context.Context, jti string, exp int64) error {
    ttl := time.Until(time.Unix(exp, 0))
    if ttl <= 0 {
        return nil // Already expired
    }
    return s.redis.SetEX(ctx, "blacklist:"+jti, "true", ttl).Err()
}

// Validate token against blacklist
func (s *TokenBlacklistService) IsTokenBlacklisted(ctx context.Context, jti string) (bool, error) {
    val, err := s.redis.Get(ctx, "blacklist:"+jti).Result()
    if err == redis.Nil {
        return false, nil
    }
    if err != nil {
        // Redis down → log warning, but allow (degraded mode)
        logger.Warn("Redis unavailable for token blacklist check", "jti", jti)
        return false, nil
    }
    return val == "true", nil
}
Middleware: AuthMiddleware (with Blacklist Check)

golang
func AuthMiddleware(blacklist *TokenBlacklistService) gin.HandlerFunc {
    return func(c *gin.Context) {
        token := c.GetHeader("Authorization")
        if token == "" { c.JSON(401, "UNAUTHORIZED"); return }
        
        claims, err := jwtService.ValidateToken(token)
        if err != nil { c.JSON(401, "UNAUTHORIZED"); return }
        
        // Check blacklist
        blacklisted, err := blacklist.IsTokenBlacklisted(c.Request.Context(), claims.Jti)
        if err == nil && blacklisted {
            c.JSON(401, "UNAUTHORIZED: Token invalidated")
            return
        }
        
        c.Set("user_id", claims.Sub)
        c.Set("user_type", claims.UserType)
        c.Next()
    }
}

func RBACMiddleware(allowedRoles ...string) gin.HandlerFunc {
    return func(c *gin.Context) {
        userType, _ := c.Get("user_type")
        allowed := false
        for _, role := range allowedRoles {
            if userType == role { allowed = true; break }
        }
        if !allowed { c.JSON(403, "FORBIDDEN"); return }
        c.Next()
    }
}

class AuthNotifier extends StateNotifier<AuthState> {
  AuthNotifier(this.apiClient) : super(const AuthState());
  
  Future<void> login(String email, String password) async {
    state = const AuthState(isLoading: true);
    try {
      final response = await apiClient.post('/auth/login', data: {
        'email': email,
        'password': password,
      });
      final token = response.data['data']['access_token'];
      await secureStorage.write(key: 'access_token', value: token);
      state = AuthState(isLoading: false, user: response.data['data']['user']);
    } catch (e) {
      state = AuthState(isLoading: false, error: e.toString());
    }
  }
  
  Future<void> logout() async {
    // Call backend blacklist endpoint
    await apiClient.post('/auth/logout');
    await secureStorage.delete(key: 'access_token');
    state = const AuthState();
  }
}
