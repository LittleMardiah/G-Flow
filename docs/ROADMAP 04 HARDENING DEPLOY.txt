FIXED v4.23 - checkLockout Redis Evaluation:

go
func (h *AdminHandler) checkLockout(ctx context.Context, adminID string) (bool, error) {
    redisCtx, cancel := context.WithTimeout(ctx, 200*time.Millisecond)
    defer cancel()
    lockoutKey := "admin:lockout:" + adminID
    redisLocked, err := h.redis.Get(redisCtx, lockoutKey).Result()
    // FIXED v4.23: Evaluasi hasil Redis secara langsung sebelum fallback
    if err == nil && redisLocked == "1" {
        return true, nil
    }
    if err != nil && !errors.Is(err, redis.Nil) {
        h.logger.Warn("Redis read failed/timeout, falling back to PostgreSQL for lockout check", "admin_id", adminID, "error", err)
    }
    var pgLocked bool
    pgErr := h.db.QueryRow(ctx, `SELECT locked_until > NOW() FROM admin_lockouts WHERE admin_id = $1`, adminID).Scan(&pgLocked)
    if pgErr == nil && pgLocked { return true, nil }
    return false, nil
}
FIXED v4.23 - UPSERT on Redis-Normal Path (No Silent Data Loss):

go
func (h *AdminHandler) recordFailedAttempt(ctx context.Context, adminID string) error {
    attemptKey := "admin:2fa_attempts:" + adminID
    var currentAttempts int64
    
    attempts, err := h.redis.Incr(ctx, attemptKey).Result()
    if err != nil {
        h.logger.Warn("Redis write failed/timeout, falling back to PostgreSQL for attempt counter", "admin_id", adminID, "error", err)
        var dbAttempts int
        errDb := h.db.QueryRow(ctx, 
            `INSERT INTO admin_lockouts (admin_id, failed_attempts, updated_at)
             VALUES ($1, 1, NOW())
             ON CONFLICT (admin_id) DO UPDATE 
             SET failed_attempts = admin_lockouts.failed_attempts + 1, updated_at = NOW() 
             RETURNING failed_attempts`, adminID).Scan(&dbAttempts)
        if errDb != nil { return errDb }
        currentAttempts = int64(dbAttempts)
    } else {
        currentAttempts = attempts
        // FIXED v4.23: UPSERT di path normal agar baris pertama admin tercipta
        h.db.Exec(ctx, 
            `INSERT INTO admin_lockouts (admin_id, failed_attempts, updated_at)
             VALUES ($1, 1, NOW())
             ON CONFLICT (admin_id) DO UPDATE 
             SET failed_attempts = admin_lockouts.failed_attempts + 1, updated_at = NOW()`, adminID)
    }

    if currentAttempts >= 3 {
        h.redis.Set(ctx, "admin:lockout:"+adminID, "1", 15*time.Minute)
        h.db.Exec(ctx, `UPDATE admin_lockouts SET locked_until = NOW() + INTERVAL '15 minutes' WHERE admin_id = $1`, adminID)
    }
    return nil
}

4.3.2 Supabase PostgreSQL & Backups — FIXED v4.23

Pool Size: 2 instances × 25 = 50. WebSocket acquires/releases per query. default_pool_size ≥ 80.
WebSocket Handler: MUST NOT retain DB connection across messages. Acquire per query, release immediately.
PgBouncer Transient Retry — SCOPE ENFORCED:

SELECT (read-only) → ✅ ALLOWED.

Mutations with Idempotency-Key → ✅ ALLOWED.

Mutations WITHOUT Idempotency-Key → ❌ FORBIDDEN.
Implementation (pseudo):
func processWithRetry(ctx, fn, isIdempotent bool) error {
if !isIdempotent && isWriteOperation(fn) { return fn() }
maxRetries := 3; base := 50ms
for i := 0; i < maxRetries; i++ {
attemptCtx, _ := context.WithTimeout(ctx, 3time.Second)
err := fn()
if err == nil { return nil }
if isTransient(err) { time.Sleep(base(1<<i) + jitter); continue }
return err
}
return errors.New("max retries exceeded")
}
