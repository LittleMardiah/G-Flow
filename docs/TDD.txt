// Handler with rate limiting
func (h *WalletHandler) TopUp(c *gin.Context) {
  // Rate limit check (Token Bucket)
  if !h.rateLimiter.Allow(c.ClientIP(), "topup", 10, time.Minute) {
    c.JSON(429, Error("Rate limit exceeded"))
    return
  }

  var req TopUpRequest
  c.BindJSON(&req)

  if req.Amount.LessThanOrEqual(decimal.Zero) || req.Amount.GreaterThan(decimal.New(100000000, 0)) {
    c.JSON(422, Error("Amount invalid"))
    return
  }

  // Idempotency
  idempotencyKey := c.GetHeader("X-Idempotency-Key")
  cached, found := h.idempotencyCache.Get(idempotencyKey)
  if found {
    c.JSON(200, cached)
    return
  }

  // Use decimal.Decimal in service
  result, err := h.walletService.TopUp(c.Request.Context(), req)
  if err != nil {
    c.JSON(500, Error(err.Error()))
    return
  }

  h.idempotencyCache.Set(idempotencyKey, result, 24*time.Hour)
  c.JSON(200, result)
}

// Service with decimal and timeout
func (s *WalletService) TopUp(ctx context.Context, req TopUpRequest) (*TopUpResponse, error) {
  tx, err := s.db.BeginTx(ctx, &sql.TxOptions{
    Isolation: sql.LevelReadCommitted,
  })
  defer tx.Rollback()

  // CRITICAL: Set statement_timeout to prevent connection exhaustion
  tx.Exec("SET LOCAL statement_timeout = '3000ms'")

  // Lock wallet
  wallet := s.repo.GetWalletForUpdate(tx, req.UserID)
  if wallet.Status != "ACTIVE" {
    return nil, errors.New("Wallet not active")
  }

  // Use decimal.Decimal for all amounts
  debitAmount := req.Amount
  creditAmount := req.Amount

  // Create ledger entries
  debitEntry := &LedgerEntry{
    WalletID:    systemEscrowID,
    EntryType:   "DEBIT",
    Amount:      debitAmount,
    ReferenceType: "TOPUP",
  }
  s.repo.CreateLedgerEntry(tx, debitEntry)

  creditEntry := &LedgerEntry{
    WalletID:    req.UserID,
    EntryType:   "CREDIT",
    Amount:      creditAmount,
    ReferenceType: "TOPUP",
  }
  s.repo.CreateLedgerEntry(tx, creditEntry)

  // Update balance
  s.repo.UpdateWalletBalance(tx, req.UserID, req.Amount)

  tx.Commit()

  return &TopUpResponse{
    TransactionID: uuid.New(),
    NewBalance:    wallet.Balance.Add(req.Amount),
  }, nil
}

// Repository with timeout
func (r *WalletRepository) GetWalletForUpdate(tx *sql.Tx, userID string) *Wallet {
  row := tx.QueryRow(`
    SELECT id, balance, status FROM wallets
    WHERE user_id = $1 FOR UPDATE
  `, userID)

  var wallet Wallet
  var balance decimal.Decimal
  row.Scan(&wallet.ID, &balance, &wallet.Status)
  wallet.Balance = balance
  return &wallet
}

// Driver location update: Write to Redis first
func (s *DriverService) UpdateLocation(ctx context.Context, driverID string, lat, lng float64) error {
  // Step 1: Write to Redis (in-memory buffer) with TTL
  err := s.redisClient.GeoAdd("driver:locations", driverID, lat, lng)
  if err != nil {
    return err
  }
  s.redisClient.Expire("driver:locations", 45*time.Second)

  // Step 2: Async persist to PostgreSQL via background worker (not blocking)
  s.taskQueue.Enqueue(&PersistLocationTask{
    DriverID: driverID,
    Lat:      lat,
    Lng:      lng,
  })

  return nil
}

// Find available drivers: Read from Redis first
func (s *DriverService) FindAvailableDrivers(ctx context.Context, lat, lng, radius float64) ([]Driver, error) {
  // Query Redis for active drivers with heartbeat
  driverIDs, err := s.redisClient.GeoRadius("driver:locations", lat, lng, radius)
  if err != nil || len(driverIDs) == 0 {
    // Fallback to PostgreSQL (but should be rare)
    return s.repo.FindDriversNearby(ctx, lat, lng, radius)
  }

  // Fetch driver details from PostgreSQL (only for active drivers)
  return s.repo.GetDriversByIDs(ctx, driverIDs)
}

// Accept order with pessimistic lock and timeout
func (s *RideService) AcceptOrder(ctx context.Context, orderID, driverID string) (*AcceptResponse, error) {
  tx, err := s.db.BeginTx(ctx, &sql.TxOptions{Isolation: sql.LevelReadCommitted})
  defer tx.Rollback()

  tx.Exec("SET LOCAL statement_timeout = '3000ms'") // CRITICAL

  order := s.repo.GetOrderForUpdate(tx, orderID)

  if order.Status != "SEARCHING_DRIVER" {
    return nil, ErrOrderNotAvailable
  }

  s.repo.UpdateOrderStatus(tx, orderID, "DRIVER_ASSIGNED", driverID)
  s.repo.CreateOrderEvent(tx, orderID, "SEARCHING_DRIVER", "DRIVER_ASSIGNED", driverID)

  tx.Commit()

  // Outbox: Publish async event for notification (decoupled)
  s.outbox.Publish(&OrderAssignedEvent{OrderID: orderID, DriverID: driverID})

  return &AcceptResponse{OrderID: orderID, Status: "DRIVER_ASSIGNED"}, nil
}

6.3 SETTLEMENT (WITH CIRCUIT BREAKER & OUTBOX)
-------------------------------------------------------------------------------
func (s *RideService) SettleOrder(ctx context.Context, orderID string) error {
  // Circuit breaker for external API calls (if any)
  err := s.circuitBreaker.Execute(func() error {
    return s.settleOrderInternal(ctx, orderID)
  })
  if err != nil {
    return err
  }

  // Outbox: Publish settlement complete event
  s.outbox.Publish(&OrderSettledEvent{OrderID: orderID})

  return nil
}

func (s *RideService) settleOrderInternal(ctx context.Context, orderID string) error {
  tx, err := s.db.BeginTx(ctx, &sql.TxOptions{Isolation: sql.LevelReadCommitted})
  defer tx.Rollback()

  tx.Exec("SET LOCAL statement_timeout = '5000ms'")

  order := s.repo.GetOrder(tx, orderID)
  fare := order.ActualFare
  commission := fare.Mul(decimal.New(10, 2).Div(decimal.New(100, 0))) // 10%
  driverEarning := fare.Sub(commission)

  // Lock wallets with ORDER BY id
  wallets := s.repo.GetWalletsForUpdate(tx, []string{
    order.CustomerWalletID,
    order.DriverWalletID,
    systemPlatformWalletID,
  })

  if wallets[0].Balance.LessThan(fare) {
    return ErrInsufficientBalance
  }

  // Create ledger entries (using decimal)
  s.repo.CreateLedgerEntry(tx, &LedgerEntry{
    WalletID:    order.CustomerWalletID,
    EntryType:   "DEBIT",
    Amount:      fare,
    ReferenceType: "RIDE_PAYMENT",
    ReferenceID: orderID,
  })
  s.repo.CreateLedgerEntry(tx, &LedgerEntry{
    WalletID:    order.DriverWalletID,
    EntryType:   "CREDIT",
    Amount:      driverEarning,
    ReferenceType: "RIDE_EARNING",
    ReferenceID: orderID,
  })
  s.repo.CreateLedgerEntry(tx, &LedgerEntry{
    WalletID:    systemPlatformWalletID,
    EntryType:   "CREDIT",
    Amount:      commission,
    ReferenceType: "RIDE_COMMISSION",
    ReferenceID: orderID,
  })

  s.repo.UpdateWalletBalance(tx, order.CustomerWalletID, fare.Neg())
  s.repo.UpdateWalletBalance(tx, order.DriverWalletID, driverEarning)
  s.repo.UpdateWalletBalance(tx, systemPlatformWalletID, commission)

  s.repo.UpdateOrderStatus(tx, orderID, "SETTLED")

  tx.Commit()
  return nil
}

// Scheduled job with Redis distributed lock
func (c *CronScheduler) RunAutoCancelOrders() {
  // Acquire lock (TTL 60s) to ensure single instance executes
  lock := c.redis.SetNX("cron:auto_cancel", "locked", 60*time.Second)

  // Execute job logic

}

      this.startPolling();
    }
  }

  startPolling() {
    // Exponential backoff: 2s, 4s, 8s, max 30s
    this.pollingInterval = setInterval(() => {
      this.fetchUpdates();
    }, this.getBackoffTime());
  }

  getBackoffTime() {
    // Exponential backoff implementation
    return Math.min(2000 * Math.pow(2, this.retryCount), 30000);
  }
}

// Server: State catch-up with sequence_id
func (h *WebSocketHandler) HandleReconnect(userID string, lastSequenceID int64) {
  // Send only events with sequence_id > lastSequenceID
  events := h.repo.GetEventsSince(userID, lastSequenceID)
  for _, event := range events {
    h.ws.WriteJSON(event)
  }
}

// Middleware
func RateLimiter(limit int, window time.Duration) gin.HandlerFunc {
  // Token bucket implementation using Redis
  return func(c *gin.Context) {
    key := "rate:" + c.ClientIP() + ":" + c.Request.URL.Path
    allowed, err := redisClient.Allow(key, limit, window)
    if err != nil || !allowed {
      c.JSON(429, Error("Rate limit exceeded"))
      c.Abort()
      return
    }
    c.Next()
  }
}

LEDGER MISMATCH ALERTING
func (s *AdminService) VerifyLedger(ctx context.Context) error {
  var debits, credits decimal.Decimal
  row := s.db.QueryRow(`
    SELECT
      SUM(CASE WHEN entry_type='DEBIT' THEN amount ELSE 0 END),
      SUM(CASE WHEN entry_type='CREDIT' THEN amount ELSE 0 END)
    FROM ledger_entries
    WHERE is_reversed = FALSE
  `)
  row.Scan(&debits, &credits)

  if !debits.Equal(credits) {
    // CRITICAL ALERT
    logger.Error("Ledger imbalance detected",
      zap.String("debits", debits.String()),
      zap.String("credits", credits.String()),
      zap.String("discrepancy", debits.Sub(credits).String()),
    )
    // Send alert to Slack/PagerDuty
    alerting.SendAlert("Ledger Imbalance", debits.Sub(credits).String())
    return errors.New("ledger imbalance")
  }
  return nil
}
