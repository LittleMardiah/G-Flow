// Test wallet balance calculation
func TestWalletBalanceAfterTopUp(t *testing.T) {
  t.Run("successful top-up", func(t *testing.T) {
    // Arrange
    wallet := &Wallet{ID: "w1", Balance: 100000}
    
    // Act
    newBalance, err := wallet.AddBalance(50000)
    
    // Assert
    assert.NoError(t, err)
    assert.Equal(t, int64(150000), newBalance)
  })
  
  t.Run("invalid amount", func(t *testing.T) {
    wallet := &Wallet{Balance: 100000}
    newBalance, err := wallet.AddBalance(-50000)
    assert.Error(t, err)
    assert.Equal(t, "Amount must be positive", err.Error())
    assert.Equal(t, int64(100000), newBalance) // unchanged
  })
}

// Test complete ride flow with settlement
func TestRideCompleteFlowWithSettlement(t *testing.T) {
  // Setup: Create test DB
  db := setupTestDB()
  defer db.Close()
  
  // Create test users
  customerID := createTestUser(db, "customer")
  driverID := createTestUser(db, "driver")
  customerWalletID := getWalletID(db, customerID)
  
  // Create initial balances
  setWalletBalance(db, customerID, 500000)
  setWalletBalance(db, driverID, 100000)
  
  // Service setup
  rideService := NewRideService(db)
  
  // Test: Customer books ride
  bookReq := BookRideRequest{
    CustomerID: customerID,
    PickupLat:  -6.2, PickupLng: 106.8,
    DropoffLat: -6.3, DropoffLng: 106.9,
  }
  orderID, err := rideService.BookRide(context.Background(), bookReq)
  assert.NoError(t, err)
  
  // Verify: Order created with SEARCHING_DRIVER status
  order, _ := rideService.GetOrder(context.Background(), orderID)
  assert.Equal(t, "SEARCHING_DRIVER", order.Status)
  assert.NotNil(t, order.EstimatedFare)
  
  // Verify: Ledger entries created (customer deducted)
  ledgerEntries := getLedgerEntries(db, customerWalletID)
  assert.True(t, len(ledgerEntries) >= 2) // Debit + Credit at least
  
  // Test: Driver accepts order
  _, err = rideService.AcceptOrder(context.Background(), orderID, driverID)
  assert.NoError(t, err)
  
  // Verify: Order status changed to DRIVER_ASSIGNED
  order, _ = rideService.GetOrder(context.Background(), orderID)
  assert.Equal(t, "DRIVER_ASSIGNED", order.Status)
  assert.Equal(t, driverID, order.DriverID)
  
  // Test: Simulate trip complete & settlement
  rideService.SettleOrder(context.Background(), orderID)
  
  // Verify: Order status SETTLED
  order, _ = rideService.GetOrder(context.Background(), orderID)
  assert.Equal(t, "SETTLED", order.Status)
  
  // Verify: Ledger balanced (debit = credit)
  balanced, _ := rideService.VerifyLedger()
  assert.True(t, balanced)
}

// Test wallet top-up API contract
func TestWalletTopUpAPI(t *testing.T) {
  prevBalance := int64(100000) // declare variable (FIXED)
  
  t.Run("successful top-up returns 200 with correct schema", func(t *testing.T) {
    // Setup
    router := setupRouter()
    w := httptest.NewRecorder()
    
    // Request
    req := httptest.NewRequest("POST", "/wallets/123/topup", 
      bytes.NewBufferString(`{"amount":500000}`))
    req.Header.Set("Authorization", "Bearer "+testJWT)
    req.Header.Set("X-Idempotency-Key", "key-123")
    
    // Execute
    router.ServeHTTP(w, req)
    
    // Assert
    assert.Equal(t, 200, w.Code)
    
    var resp TopUpResponse
    json.Unmarshal(w.Body.Bytes(), &resp)
    assert.NotEmpty(t, resp.TransactionID)
    assert.Equal(t, prevBalance + 500000, resp.NewBalance)
  })
  
  t.Run("missing idempotency key returns 400", func(t *testing.T) {
    router := setupRouter()
    w := httptest.NewRecorder()
    
    req := httptest.NewRequest("POST", "/wallets/123/topup", 
      bytes.NewBufferString(`{"amount":500000}`))
    req.Header.Set("Authorization", "Bearer "+testJWT)
    // Missing X-Idempotency-Key
    
    router.ServeHTTP(w, req)
    
    assert.Equal(t, 400, w.Code)
  })
}

export default function () {
  const payload = JSON.stringify({
    to_wallet_id: 'wallet-b',

import http from 'k6/http';
import { check, sleep } from 'k6';

// --- FIXED: testJWT defined via __ENV ---
const testJWT = __ENV.JWT_TOKEN || 'test-jwt-token-for-load-test';

export let options = {
  stages: [
    { duration: '2m', target: 50 },
    { duration: '8m', target: 50 },
    { duration: '2m', target: 0 },
  ],
  thresholds: {
    http_req_duration: ['p(95)<200', 'p(99)<300'],
    http_req_failed: ['rate<0.001'],
  },
};

export default function () {
  const payload = JSON.stringify({
    to_wallet_id: 'wallet-b',
    amount: 50000,
  });
  
  const response = http.post('http://localhost:8080/wallets/transfer', payload, {
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${testJWT}`,
      'X-Idempotency-Key': `key-${__VU}-${__ITER}`,
    },
  });
  
  check(response, {
    'status is 200': (r) => r.status === 200,
    'latency p95 < 200ms': (r) => r.timings.duration < 200,
    'latency p99 < 300ms': (r) => r.timings.duration < 300,
    'body has transaction_id': (r) => r.json('transaction_id') !== undefined,
  });
  
  sleep(1);
}

// --- FIXED: teardown function added ---
export function teardown() {
  // Reset test data sandbox

    headers: {
      'Authorization': `Bearer ${testJWT}`,
      'Content-Type': 'application/json',
    },
  });
}
